Q1. How are Kanto region Pokémon divided among growth rates, and which Pokémon have the fastest growth rate?

# query total number of pokemon for each growth rate
q1.1_query <- dbGetQuery(kanto_database,
                         "SELECT growth_rate, COUNT(*) AS total
                         FROM training_breeding_stats
                         LEFT JOIN growth_rate
                            ON training_breeding_stats.growth_rate_id = growth_rate.growth_rate_id
                         GROUP BY growth_rate
                         ORDER BY total DESC;")

# query pokemon with a fast growth rate descending by total base stats
q1.2_query <- dbGetQuery(kanto_database,
                         "SELECT training_breeding_stats.national_no, training_breeding_stats.pokemon,
                                 type_1, hp, attack, defense, sp_atk, sp_def, speed, total
                         FROM training_breeding_stats
                         LEFT JOIN growth_rate
                            ON training_breeding_stats.growth_rate_id = growth_rate.growth_rate_id
                         LEFT JOIN base_stats
                            ON training_breeding_stats.national_no = base_stats.national_no
                         WHERE growth_rate.growth_rate LIKE '%Fast%'
                         ORDER BY total DESC;")

Q2. How are Kanto region Pokémon divided among egg groups? Are there any noticeable trends in either egg group 1 or egg group 2?

# query total number of pokemon for each egg group (1 and 2)
q2_query <- dbGetQuery(kanto_database,
                       "SELECT egg_group_id, egg_group, egg_1_total, egg_2_total
                       FROM egg_groups
                       LEFT JOIN (SELECT egg_group_1, count(*) AS egg_1_total
                                  FROM egg_info
                                  GROUP BY egg_group_1) AS table_1
                          ON egg_groups.egg_group_id = table_1.egg_group_1
                       LEFT JOIN (SELECT egg_group_2, count(*) AS egg_2_total
                                  FROM egg_info
                                  GROUP BY egg_group_2) AS table_2
                          ON egg_groups.egg_group_id = table_2.egg_group_2
                       ORDER BY egg_group_id ASC;")
